def sha(raw):return hashlib.sha256(raw).hexdigest()
def encode(value):return (json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()
def check():

def read(path,expected=None,size=None):
 p=pathlib.Path(path);s=p.lstat();assert p.is_absolute() and p==p.resolve() and p.is_file() and not s.st_flags&0x40000000

 assert expected is None or sha(raw)==expected
 sources[str(p)]=dict(bytes=len(raw),sha256=sha(raw));return raw
