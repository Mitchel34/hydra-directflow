def equal(actual,expected):
 global scalars,maxabs,maxscaled
 if expected is None:assert actual is None;return
 if isinstance(expected,(bool,str,int,np.integer)):
  assert actual==expected,(actual,expected);return
 assert isinstance(actual,(int,float)) and math.isfinite(actual) and math.isfinite(float(expected)),(actual,expected)
 expected=float(expected);diff=abs(actual-expected);scale=ATOL+RTOL*abs(expected)
 maxabs=max(maxabs,diff);maxscaled=max(maxscaled,diff/scale);scalars+=1
 assert diff<=scale,(actual,expected,diff,scale)
